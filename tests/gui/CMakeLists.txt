cmake_minimum_required(VERSION 3.16)
project(GuiAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Test Widgets Charts Network Concurrent)
include("${AUDIT_DIR}/objects.cmake")
add_executable(tst_gui_functional tst_gui_functional.cpp ${APP_OBJECTS})
file(GLOB_RECURSE APP_HEADERS "${REPO}/src/*.h")
foreach(header ${APP_HEADERS})
  get_filename_component(dir "${header}" DIRECTORY)
  list(APPEND APP_INCLUDES "${dir}")
endforeach()
list(REMOVE_DUPLICATES APP_INCLUDES)
target_include_directories(tst_gui_functional PRIVATE ${APP_INCLUDES})
target_link_libraries(tst_gui_functional PRIVATE Qt6::Test Qt6::Widgets Qt6::Charts Qt6::Network Qt6::Concurrent)
target_compile_definitions(tst_gui_functional PRIVATE QT_NO_DEBUG QT_NO_DEBUG_OUTPUT REPO="${REPO}" AUDIT_DIR="${AUDIT_DIR}" APP_BUILD="${APP_BUILD}")
target_compile_options(tst_gui_functional PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
