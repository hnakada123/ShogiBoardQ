cmake_minimum_required(VERSION 3.16)
project(GuiAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Test Widgets Charts Network Concurrent Multimedia Sql)
include("${AUDIT_DIR}/objects.cmake")
file(GLOB_RECURSE APP_HEADERS "${REPO}/src/*.h")
foreach(header ${APP_HEADERS})
  get_filename_component(dir "${header}" DIRECTORY)
  list(APPEND APP_INCLUDES "${dir}")
endforeach()
list(REMOVE_DUPLICATES APP_INCLUDES)
# アプリを P2P対局を含めてビルドしたとき（ENABLE_P2P_PLAY）は、アプリと同じ定義とライブラリ（libsodium・QtKeychain）で組む
load_cache("${APP_BUILD}" READ_WITH_PREFIX APP_ ENABLE_P2P_PLAY Qt6Keychain_DIR)
set(P2P_LIBS)
set(P2P_DEFINITIONS)
if(APP_ENABLE_P2P_PLAY)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium>=1.0.18)
  list(APPEND P2P_LIBS PkgConfig::SODIUM)
  list(APPEND P2P_DEFINITIONS SBQ_ENABLE_P2P=1)
  if(APP_Qt6Keychain_DIR)
    find_package(Qt6Keychain CONFIG REQUIRED PATHS "${APP_Qt6Keychain_DIR}" NO_DEFAULT_PATH)
    list(APPEND P2P_LIBS Qt6Keychain::Qt6Keychain)
    list(APPEND P2P_DEFINITIONS SBQ_HAVE_QTKEYCHAIN=1)
  endif()
endif()
add_library(hayanagi_tsume STATIC IMPORTED)
set_target_properties(hayanagi_tsume PROPERTIES
  IMPORTED_LOCATION "${APP_BUILD}/Hayanagi/libhayanagi_tsume.a"
  INTERFACE_INCLUDE_DIRECTORIES "${REPO}/Hayanagi/src")
foreach(test IN ITEMS tst_gui_functional tst_start_game_flow tst_ponder_flow tst_tsume_play_gui tst_tsume_collection_gui tst_tsumeshogi_generator_gui tst_tsume_search_dialog engine_shutdown_harness)
  add_executable(${test} ${test}.cpp ${APP_OBJECTS})
  target_include_directories(${test} PRIVATE ${APP_INCLUDES})
  target_link_libraries(${test} PRIVATE hayanagi_tsume Qt6::Test Qt6::Widgets Qt6::Charts Qt6::Network Qt6::Concurrent Qt6::Multimedia Qt6::Sql ${P2P_LIBS})
  target_compile_definitions(${test} PRIVATE QT_NO_DEBUG QT_NO_DEBUG_OUTPUT REPO="${REPO}" AUDIT_DIR="${AUDIT_DIR}" APP_BUILD="${APP_BUILD}" ${P2P_DEFINITIONS})
  target_compile_options(${test} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
endforeach()
