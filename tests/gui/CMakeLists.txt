cmake_minimum_required(VERSION 3.16)
project(GuiAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Test Widgets Charts Network Concurrent Multimedia)
include("${AUDIT_DIR}/objects.cmake")
file(GLOB_RECURSE APP_HEADERS "${REPO}/src/*.h")
foreach(header ${APP_HEADERS})
  get_filename_component(dir "${header}" DIRECTORY)
  list(APPEND APP_INCLUDES "${dir}")
endforeach()
list(REMOVE_DUPLICATES APP_INCLUDES)
add_library(hayanagi_tsume STATIC IMPORTED)
set_target_properties(hayanagi_tsume PROPERTIES
  IMPORTED_LOCATION "${APP_BUILD}/Hayanagi/libhayanagi_tsume.a"
  INTERFACE_INCLUDE_DIRECTORIES "${REPO}/Hayanagi/src")
foreach(test IN ITEMS tst_gui_functional tst_start_game_flow tst_ponder_flow tst_tsume_play_gui)
  add_executable(${test} ${test}.cpp ${APP_OBJECTS})
  target_include_directories(${test} PRIVATE ${APP_INCLUDES})
  target_link_libraries(${test} PRIVATE hayanagi_tsume Qt6::Test Qt6::Widgets Qt6::Charts Qt6::Network Qt6::Concurrent Qt6::Multimedia)
  target_compile_definitions(${test} PRIVATE QT_NO_DEBUG QT_NO_DEBUG_OUTPUT REPO="${REPO}" AUDIT_DIR="${AUDIT_DIR}" APP_BUILD="${APP_BUILD}")
  target_compile_options(${test} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
endforeach()
